abstract class BillingStrategy {
    + {abstract} calculateBill(charges : List<Charge>) : double
}

class InsuranceBilling extends BillingStrategy {
    - coveragePercentage : double

    + calculateBill(charges : List<Charge>) : double
}

class GovernmentBilling extends BillingStrategy {
    - subsidizedRate : double
    + calculateBill(charges : List<Charge>) : double
}

class SelfPayBilling extends BillingStrategy {
    + calculateBill(charges : List<Charge>) : double
}

    - name : string
    - admissionDate : date
    - dischargeDate : date
    + getId() : int
    + getName() : string

    + getDischargeDate() : date
}

class DischargeBill {
    - billId : int
    - patient : Patient

    - totalAmount : double
    + generateBill()

}

abstract class Charge {
    - chargeId : int
    - description : string
    - amount : double
    + {abstract} calculateAmount() : double
    + getDescription() : string
    + getAmount() : double
}

class ProcedureCharge extends Charge {
    - procedureName : string
    + calculateAmount() : double
}

class MedicationCharge extends Charge {
    - medicationName : string
    - quantity : int
    + calculateAmount() : double
}

class DiagnosticCharge extends Charge {
    - testName : string
    + calculateAmount() : double
}

class RoomCharge extends Charge {
    - roomType : string
    - days : int
    + calculateAmount() : double
}

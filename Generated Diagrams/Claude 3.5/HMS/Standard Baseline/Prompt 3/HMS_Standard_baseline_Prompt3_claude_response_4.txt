  - coveragePercentage: double

}

  - subsidyAmount: double

}

class SelfPayBillingStrategy {
  + calculateBill(patient: Patient, charges: List<Charge>): Invoice
}

class Patient {
  - patientId: String
  - name: String
  - category: PatientCategory
  - billingStrategy: BillingStrategy
  + setBillingStrategy(strategy: BillingStrategy): void

}

enum PatientCategory {
  INSURED
  GOVERNMENT_BENEFICIARY
  SELF_PAY
}

abstract class Charge {
  # chargeId: String
  # description: String
  # baseAmount: double
  + {abstract} calculateAmount(): double
  + getDescription(): String
  + getBaseAmount(): double
}

  - procedureCode: String

}

class MedicationCharge {
  - medicationName: String
  - quantity: int
  - unitPrice: double
  + calculateAmount(): double
}

class DiagnosticServiceCharge {
  - serviceType: String

  + calculateAmount(): double
}

class AccommodationCharge {
  - roomType: String

  - dailyRate: double

}

class Invoice {

  - coveredAmount: double
  - patientResponsibility: double

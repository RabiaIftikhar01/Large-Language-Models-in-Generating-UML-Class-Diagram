class WeatherStation {
    +id: String
    +location: String
    +sensorData: List<SensorData>

}

class Sensor {
    +id: String
    +type: String
    +unit: String
    +getData(): SensorData
}

class SensorData {
    +timestamp: Date
    +value: float
    +format: String
}

class WeatherInformationSystem {
    +stations: List<WeatherStation>
    +ingestData(station: WeatherStation): void
    +normalizeData(data: SensorData): NormalizedData
}

class NormalizedData {
    +timestamp: Date
    +value: float
    +unit: String
}

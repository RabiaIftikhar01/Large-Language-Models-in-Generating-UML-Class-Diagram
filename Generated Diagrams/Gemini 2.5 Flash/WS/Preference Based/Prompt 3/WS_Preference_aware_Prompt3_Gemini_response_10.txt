' Define the classes for the environmental monitoring system

class Agency {
    - agencyId : String
    - name : String
    - type : String ' e.g., Public, Research, Infrastructure
    + getAgencyId() : String
    + getName() : String
}

class WeatherStation {
    - stationId : String
    - name : String
    - location : String
    - transmissionCadence : Duration ' e.g., hourly, 30min
    + getStationId() : String
    + getName() : String
    + requestRawSensorReadings() : List<RawSensorReading>
}

class Sensor {
    - sensorId : String
    - type : String ' e.g., Temperature, Humidity, WindSpeed
    - vendorName : String
    - nativeFormat : String ' e.g., JSON, XML, proprietary binary
    + getSensorId() : String
    + getType() : String
    + generateRawReading() : RawSensorReading
}

class RawSensorReading {
    - readingId : String
    - sensorId : String
    - stationId : String
    - timestamp : DateTime
    - rawValue : String ' The raw data string/blob
    - nativeFormat : String
    + getReadingId() : String
    + getSensorId() : String
    + getStationId() : String
    + getTimestamp() : DateTime
    + getRawValue() : String
    + getNativeFormat() : String
}

class SensorData {
    - dataId : String
    - stationId : String
    - sensorId : String
    - timestamp : DateTime
    - value : Double ' Standardized numeric value
    - unit : String ' e.g., Celsius, %, m/s
    + getDataId() : String
    + getStationId() : String
    + getSensorId() : String
    + getTimestamp() : DateTime
    + getValue() : Double
    + getUnit() : String
}

class DataIngestionService {
    + receiveRawData(reading : RawSensorReading) : void
    + transformToStandard(rawReading : RawSensorReading) : SensorData
    + processStandardizedData(data : SensorData) : void
}

class DataStorage {
    + saveSensorData(data : SensorData) : void
    + retrieveSensorData(query : String) : List<SensorData>
}

' Define relationships between classes

' An Agency operates one or more WeatherStations
Agency "1" -- "*" WeatherStation : operates >

' A WeatherStation contains one or more Sensors (composition)
WeatherStation "1" *-- "1..*" Sensor : contains >

' A WeatherStation creates and sends RawSensorReadings
WeatherStation ..> RawSensorReading : creates & sends >

' A Sensor generates RawSensorReadings
Sensor ..> RawSensorReading : generates >

' The DataIngestionService receives RawSensorReadings
RawSensorReading <.. DataIngestionService : received by <

' The DataIngestionService transforms RawSensorReadings into SensorData
RawSensorReading <.. DataIngestionService : transforms <
DataIngestionService ..> SensorData : creates >

' The DataIngestionService stores SensorData in DataStorage
DataIngestionService --> DataStorage : stores >

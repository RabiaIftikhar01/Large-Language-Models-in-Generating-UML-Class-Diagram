' Title for the diagram
title Environmental Monitoring System - Class Diagram

' Main System Component
class MonitoringSystem {
    - id : String
    - name : String
    + requestDataTransfer(stationId : String) : void
    + ingestData(telemetryData : TelemetryData) : void
    + processStandardizedData(data : TelemetryData) : void
}

' Service for handling data ingestion and standardization
class DataIngestionService {
    - dataStandardizer : DataStandardizer
    + receiveRawData(rawData : TelemetryData) : TelemetryData
    + standardize(rawData : TelemetryData) : TelemetryData
}

' Interface for data standardization strategy
interface DataStandardizer {
    + standardize(rawData : TelemetryData) : TelemetryData
}

' Concrete implementation for data standardization
class GenericDataStandardizer implements DataStandardizer {
    + standardize(rawData : TelemetryData) : TelemetryData
}

' Represents a physical weather station
class WeatherStation {
    - id : String
    - location : String
    - ownerType : String ' e.g., "Public Agency", "Research Center", "Infrastructure Entity"
    - transferCadence : String ' e.g., "hourly", "daily", "minutely"
    + transmitData() : List<TelemetryData>
    + getStationInfo() : String
}

' Represents a sensor within a weather station
class Sensor {
    - id : String
    - type : String ' e.g., "temperature", "humidity", "pressure"
    - vendor : String
    - nativeFormat : String ' e.g., "JSON", "XML", "CSV"
    + collectRawData() : String ' Returns raw data string
    + getSensorInfo() : String
}

' Represents the telemetry data collected by a sensor
class TelemetryData {
    - id : String
    - timestamp : DateTime
    - rawValue : String
    - standardizedValue : String
    - sensorId : String
    - stationId : String
    + getRawValue() : String
    + getStandardizedValue() : String
    + updateStandardizedValue(value : String) : void
}

' Classes representing key entities and components
class EnvironmentalMonitoringPlatform {
    - platformId : String
    - name : String
    -- Operations --
    + ingestRawData(rawData : RawMeasurement)
    + storeNormalizedData(data : NormalizedMeasurement)
    + monitorStations()
}

class WeatherInformationSystem {
    - systemId : String
    - name : String
    -- Operations --
    + establishLink(stationId : String)
    + requestDataTransmission(station : WeatherStation)
    + receiveRawData(rawData : RawMeasurement)
}

class WeatherStation {
    - stationId : String
    - location : String
    - lastTransmissionTime : DateTime
    - transmissionFrequencyHours : int
    -- Operations --
    + transmitData() : RawMeasurement
    + updateTransmissionFrequency(hours : int)
    + getStatus() : String
}

class Sensor {
    - sensorId : String
    - manufacturer : String
    - model : String
    - type : String ' e.g., Temperature, Humidity, Pressure
    - nativeFormatDescription : String
    -- Operations --
    + readNativeData() : String
    + calibrate()
}

class RawMeasurement {
    - measurementId : String
    - timestamp : DateTime
    - sensorId : String
    - nativeValue : String ' Raw string from sensor
    - nativeUnit : String
    -- Operations --
    + getNativeData() : String
}

class NormalizedMeasurement {
    - measurementId : String
    - timestamp : DateTime
    - sensorId : String
    - value : double
    - unit : String ' Standardized unit
    -- Operations --
    + getValue() : double
    + getUnit() : String
}

class DataNormalizer {
    -- Operations --
    + normalize(raw : RawMeasurement) : NormalizedMeasurement
}

class Organization {
    - organizationId : String
    - name : String
    - type : String ' e.g., Government Agency, Research Institution, Infrastructure Provider
    -- Operations --
    + registerStation(station : WeatherStation)
}

' Relationships between classes

' The EnvironmentalMonitoringPlatform manages a WeatherInformationSystem component.
EnvironmentalMonitoringPlatform "1" *-- "1" WeatherInformationSystem : manages >

' The WeatherInformationSystem requests and receives data from multiple WeatherStations.
WeatherInformationSystem "1" --> "1..*" WeatherStation : requests/receives data from >

' A WeatherStation hosts multiple Sensor devices.
WeatherStation "1" o-- "1..*" Sensor : hosts

' A Sensor produces RawMeasurements.
Sensor "1" --> "0..*" RawMeasurement : produces >

' A RawMeasurement is processed by a DataNormalizer.
RawMeasurement "1" --> "1" DataNormalizer : is processed by >

' A DataNormalizer creates a NormalizedMeasurement from a RawMeasurement.
DataNormalizer "1" --> "1" NormalizedMeasurement : creates >

' The EnvironmentalMonitoringPlatform stores the NormalizedMeasurements.
EnvironmentalMonitoringPlatform "1" <-- "0..*" NormalizedMeasurement : stores

' An Organization operates multiple WeatherStations.
Organization "1" -- "0..*" WeatherStation : operates >

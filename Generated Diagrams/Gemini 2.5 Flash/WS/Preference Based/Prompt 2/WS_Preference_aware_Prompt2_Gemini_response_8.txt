' Main orchestrating system for environmental monitoring
class EnvironmentalMonitoringSystem {
    - id : String
    + requestTelemetry(stationId : String) : void
    + handleRawTelemetry(stationId : String, rawDataList : List<RawSensorData>) : void
}

' Represents raw data from a sensor before normalization
class RawSensorData {
    - sensorId : String
    - rawValue : String
    - proprietaryFormatSpec : String
    - timestamp : DateTime
}

' Handles high-volume data ingestion and processing, including normalization
class DataIngestor {
    - id : String
    + ingest(stationId : String, rawDataList : List<RawSensorData>) : List<SensorReading>
}

' Component responsible for normalizing proprietary sensor data formats
class DataNormalizer {
    + normalize(rawData : String, formatSpec : String) : double
}

' Represents a physical weather station collecting telemetry
class WeatherStation {
    - id : String
    - location : String
    - geographicRegion : String
    - reportingInterval : Duration
    + collectRawReadings() : List<RawSensorData>
}

' Represents an individual sensor within a weather station
class Sensor {
    - id : String
    - type : String
    - proprietaryFormatSpec : String
    + readRawValue() : String
}

' Represents a single, normalized measurement from a sensor
class SensorReading {
    - id : String
    - sensorId : String
    - timestamp : DateTime
    - rawValue : String
    - normalizedValue : double
    - unit : String
}

' Represents an entity (government, academic, operator) managing weather stations
class Organization {
    - id : String
    - name : String
    - type : String ' e.g., Government, Academic, Infrastructure
    + addStation(station : WeatherStation) : void
    + removeStation(station : WeatherStation) : void
}

' Represents a manufacturer of sensors
class Manufacturer {
    - id : String
    - name : String
}

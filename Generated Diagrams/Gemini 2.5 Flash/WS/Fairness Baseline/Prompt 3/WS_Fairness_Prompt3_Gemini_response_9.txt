' UML Class Diagram for an Environmental Monitoring System
'
' Design Principles Applied:
' - Simplicity and Readability: Focuses on core entities and their essential interactions.
' - Consistent Naming: Classes are singular nouns (PascalCase), attributes and operations are camelCase.
' - Attribute and Operation Assignment: Carefully placed attributes and operations to appropriate classes, ensuring a manageable set of responsibilities.
' - Access Specifiers: Private attributes (`-`), public operations (`+`).
' - Relationships: Utilizes aggregation, association, and dependency to model interactions.
' - Multiplicity: Clearly defined for all associations to reflect business rules.
' - Unidirectional Associations: Used where one class needs to know about another, promoting loose coupling.
' - Descriptive Labels: Relationship names clarify the purpose of connections.
' - No Inheritance or Circular Dependencies: Adheres to best practices for maintainability.
' - Explanatory Notes: Provides additional context for complex elements.

' Classes
class Organization {
  - organizationId: String
  - name: String
  - type: String ' e.g., "PublicAgency", "ResearchCenter", "InfrastructureEntity"
}

class Vendor {
  - vendorId: String
  - name: String
}

class WeatherStation {
  - stationId: String
  - name: String
  - location: String
  - dataCadence: String ' e.g., "hourly", "daily", "custom"
  --
  + getRawSensorReadings(): List<SensorReading>
}

class Sensor {
  - sensorId: String
  - type: String ' e.g., "Temperature", "Humidity", "Pressure"
  - unit: String ' e.g., "Celsius", "RelativeHumidity", "hPa"
  - model: String
  --
  + captureReading(): SensorReading
}

class SensorReading {
  - readingId: String
  - timestamp: DateTime
  - rawValue: Double
  - standardizedValue: Double
  - isStandardized: Boolean = false
}

class DataIngestionService {
  --
  + ingestDataFromStation(station: WeatherStation): void
  + processRawReading(reading: SensorReading): SensorReading
  + storeReading(reading: SensorReading): void
}

' Relationships

' An Organization owns multiple WeatherStations.
' Aggregation indicates that WeatherStations can exist independently of an Organization,
' but are managed by one.
Organization "1" o-- "0..*" WeatherStation : owns

' A WeatherStation is equipped with multiple Sensors.
' Aggregation indicates that Sensors are part of a WeatherStation,
' but could potentially be moved or replaced.
WeatherStation "1" o-- "1..*" Sensor : contains

' A Sensor is supplied by a Vendor.
' Unidirectional association from Sensor to Vendor implies that a Sensor knows its Vendor,
' but the Vendor may not directly manage a collection of all Sensors it has supplied.
Sensor "*" --> "1" Vendor : supplied by

' A SensorReading is generated by a Sensor.
' Unidirectional association from SensorReading to Sensor implies that a SensorReading
' knows its source Sensor for context and processing.
SensorReading "*" --> "1" Sensor : from

' DataIngestionService depends on WeatherStation to request data.
DataIngestionService ..> WeatherStation : <<requests data from>>

' DataIngestionService depends on SensorReading to process and store the data.
DataIngestionService ..> SensorReading : <<processes & stores>>

' Notes for additional clarity
note top of DataIngestionService
  Orchestrates data collection from stations,
  standardizes raw sensor data, and persists it.
end note

note right of SensorReading
  'standardizedValue' is populated
  by the DataIngestionService.
  'isStandardized' tracks processing status.
end note

@enduml

    BackgroundColor White

skinparam note {
    BackgroundColor #FFFFCC
    BorderColor #DAA520

' Classes are named as singular nouns, short, and in PascalCase.
' Attributes are private (indicated by -) and methods are public (indicated by +) unless internal helper methods (private).
' Good practice dictates accessors (get/set) for private attributes, which are implied here for brevity.

class Organization {
    - id: String
    - name: String
    - type: String ' e.g., "Public Agency", "Research Center"
    --
    + getOrganizationInfo(): String
}

class Station {
    - id: String
    - location: String
    - dataCadence: Duration ' Hourly, daily, etc.
    --
    + requestDataTransfer(): RawSensorData[]
    + configureCadence(newCadence: Duration)
    + getStationInfo(): String
}

class Sensor {
    - id: String
    - type: String ' e.g., "Temperature", "Humidity", "Pressure"
    - model: String
    - nativeFormat: String ' e.g., "JSON", "XML", "CSV"
    --
    + readRawData(): RawSensorData
    + getSensorInfo(): String
}

class Vendor {
    - id: String
    - name: String
    --
    + getVendorInfo(): String
}

class RawSensorData {
    - id: String
    - payload: String ' The data in its original, native format
    - timestamp: DateTime
    --
    + getPayload(): String
    + getTimestamp(): DateTime
}

class StandardizedSensorData {
    - id: String
    - value: double
    - unit: String ' e.g., "Celsius", "hPa", "%"
    - timestamp: DateTime
    - sensorType: String ' Redundant but useful for quick lookup without navigating to Sensor
    --
    + getValue(): double
    + getUnit(): String
    + getTimestamp(): DateTime
    + getSensorType(): String
}

class DataIngestionService {
    --
    + ingest(rawData: RawSensorData): StandardizedSensorData
    - standardize(rawData: RawSensorData): StandardizedSensorData
    - storeData(standardizedData: StandardizedSensorData): void
}

' Relationships with Multiplicity and Role Names
' Unidirectional associations are preferred for reduced coupling and improved reusability.

Organization "1" -- "0..*" Station : operates >
note on link
  An Organization operates zero or many Stations.
  A Station is operated by exactly one Organization.
end note

Station "1" -- "1..*" Sensor : has >
note on link
  A Station has one or many Sensors.
  A Sensor belongs to exactly one Station.
end note

Sensor "0..*" -- "1" Vendor : manufactured by >
note on link
  A Sensor is manufactured by one Vendor.
  A Vendor manufactures zero or many Sensors.
  (Unidirectional: Sensor knows its Vendor)
end note

Station "1" -- "0..*" RawSensorData : transmits >
note on link
  A Station transmits zero or many RawSensorData records.
  A RawSensorData record is transmitted by exactly one Station.
end note

RawSensorData "0..*" -- "1" Sensor : generated by >
note on link
  RawSensorData is generated by a specific Sensor.
  (Unidirectional: RawSensorData knows its generating Sensor)
end note

DataIngestionService "1" --> "1..*" Station : requestsDataFrom >
note on link
  The DataIngestionService requests data from one or many Stations.
  (Unidirectional: Service initiates communication with Station)
end note

DataIngestionService "1" -- "0..*" RawSensorData : receives >
note on link
  The DataIngestionService receives zero or many RawSensorData records.
end note

DataIngestionService "1" -- "0..*" StandardizedSensorData : produces >
note on link
  The DataIngestionService produces zero or many StandardizedSensorData records.
end note

StandardizedSensorData "0..*" -- "1" Sensor : derivedFrom >
note on link
  StandardizedSensorData is derived from a specific Sensor.
  (Unidirectional: StandardizedSensorData knows its source Sensor)
end note

' Explanatory Notes
note top of DataIngestionService
  This service is responsible for
  connecting to stations, receiving
  raw data, standardizing it, and
  persisting the standardized data.
end note

note top of RawSensorData
  Represents sensor telemetry
  in its original, native format
  as reported by the device.
end note

note top of StandardizedSensorData
  Represents sensor telemetry
  after conversion to a common,
  system-wide standardized format.
end note

@enduml

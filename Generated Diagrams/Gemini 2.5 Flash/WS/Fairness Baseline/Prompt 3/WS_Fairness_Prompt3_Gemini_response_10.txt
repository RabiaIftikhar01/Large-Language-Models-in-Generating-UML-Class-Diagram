' Define the classes
class MonitoringSystem {
  - systemId: String
  - name: String
  --
  + MonitoringSystem(systemId: String, name: String)
  + requestTelemetry(stationId: String): List<SensorData>
  + processRawTelemetry(rawData: List<SensorData>): List<SensorData>
  + storeTelemetry(standardizedData: List<SensorData>): void
}

class WeatherStation {
  - stationId: String
  - name: String
  - location: String
  - latitude: double
  - longitude: double
  - transmissionCadence: String ' e.g., "hourly", "daily", "custom"
  --
  + WeatherStation(stationId: String, name: String, location: String, latitude: double, longitude: double, cadence: String)
  + collectDataFromSensors(): List<SensorData>
  + transmitData(): List<SensorData>
}

class Sensor {
  - sensorId: String
  - type: String ' e.g., "temperature", "humidity", "pressure"
  - unit: String ' e.g., "Celsius", "%", "hPa"
  - vendor: String
  - model: String
  --
  + Sensor(sensorId: String, type: String, unit: String, vendor: String, model: String)
  + readMeasurement(): SensorData
}

class SensorData {
  - dataId: String
  - timestamp: DateTime
  - value: double
  - unit: String
  - nativeFormat: String ' Original format reported by sensor
  - isStandardized: boolean = false
  --
  + SensorData(dataId: String, timestamp: DateTime, value: double, unit: String, nativeFormat: String)
  + getValue(): double
  + getTimestamp(): DateTime
  + getNativeFormat(): String
  + setStandardized(isStandardized: boolean): void
}

class Organization {
  - orgId: String
  - name: String
  - type: String ' e.g., "PublicAgency", "ResearchCenter", "InfrastructureEntity"
  --
  + Organization(orgId: String, name: String, type: String)
  + getName(): String
  + getType(): String
}

' Define relationships between classes
MonitoringSystem "1" -- "0..*" WeatherStation : requests data from >
' The MonitoringSystem connects to and requests data from multiple WeatherStations.

Organization "1" -- "1..*" WeatherStation : operates >
' An Organization operates one or more WeatherStations.
' A WeatherStation must be operated by exactly one Organization.

WeatherStation "1" *-- "1..*" Sensor : contains
' A WeatherStation contains one or more Sensors.
' Sensors are a part of a WeatherStation (composition).

Sensor "1" -- "0..*" SensorData : generates >
' A Sensor generates zero or more SensorData instances over time.
' Each SensorData instance is generated by exactly one Sensor.

' Add explanatory notes for clarity
note top of MonitoringSystem
  The central component responsible for
  data ingestion, standardization, and storage.
  Handles large-scale telemetry processing.
end note

note top of SensorData
  Represents a single data point reported by a sensor.
  Includes its native format and a flag for standardization.
end note

note right of WeatherStation
  Represents a physical weather station.
  Manages its sensors and data transmission.
end note

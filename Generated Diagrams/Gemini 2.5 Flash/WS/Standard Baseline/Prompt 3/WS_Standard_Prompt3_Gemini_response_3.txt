title Environmental Monitoring System - Class Diagram

class EnvironmentalMonitoringSystem {
    - name: String
    - architectureType: String = "Distributed"
    - ingestionCapacity: String = "Large-scale"
    - reliabilityLevel: String = "High"
    + requestDataTransfer(stationId: String): void
    + ingestRawData(rawData: TelemetryData): void
    + standardizeData(rawData: TelemetryData): TelemetryData
    + storeStandardizedData(standardizedData: TelemetryData): void
}

class WeatherStation {
    - stationId: String
    - location: String
    - ownerOrganization: String // e.g., Public Agency, Research Center, Infrastructure Entity
    - dataCadence: String = "Hourly" // Subject to future adjustment
    + transmitDataBatch(): TelemetryData[]
}

class Sensor {
    - sensorId: String
    - sensorType: String // e.g., "Temperature", "Humidity", "Wind Speed"
    - model: String
    + readRawData(): TelemetryData // Reads a single data point
}

class TelemetryData {
    - timestamp: DateTime
    - rawValue: String
    - rawUnit: String
    - nativeFormat: String // e.g., XML, JSON, proprietary binary
    - standardizedValue: String
    - standardizedUnit: String
    - standardizedFormat: String // e.g., JSON-LD, GeoJSON
}

class SensorVendor {
    - vendorName: String
    - contactInfo: String
}

' Relationships

EnvironmentalMonitoringSystem "1" -- "*" WeatherStation : monitors / connects to >
WeatherStation "1" *-- "1..*" Sensor : contains >
Sensor "1" -- "0..*" TelemetryData : generates >

' A TelemetryData point is associated with the Sensor that generated it
TelemetryData "0..*" -- "1" Sensor : generated by >

' A Sensor is manufactured by a Vendor
Sensor "0..*" -- "1" SensorVendor : manufactured by >

' Notes for clarity
note "Handles large-scale ingestion\nand standardization of data" as N_EMS
EnvironmentalMonitoringSystem .. N_EMS

note "Reports data in native formats\nwhich are then standardized" as N_TD
TelemetryData .. N_TD

note "Data cadence can vary by station\nand is configurable" as N_WS
WeatherStation .. N_WS

@enduml

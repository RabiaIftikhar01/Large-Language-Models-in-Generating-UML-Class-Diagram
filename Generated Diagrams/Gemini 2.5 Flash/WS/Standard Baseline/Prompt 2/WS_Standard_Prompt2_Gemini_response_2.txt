title Environmental Monitoring Platform - Class Diagram

class EnvironmentalMonitoringPlatform {
  -platformId: String
  -name: String
  +accommodateHighVolumeIngestion()
  +maintainDependableOperationAcrossRegions()
  +initiateCommunication(station: WeatherStation)
  +requestDataTransmission(station: WeatherStation)
}

class WeatherStation {
  -stationId: String
  -location: String // e.g., "Lat: X, Lon: Y"
  -reportingInterval: Duration // e.g., 1 hour, 30 minutes
  +transmitData(): List<Measurement>
}

class Sensor {
  -sensorId: String
  -sensorType: String // e.g., "Temperature", "Humidity", "Pressure"
  -proprietaryFormatDetails: String // e.g., "VendorX_FormatV2_JSON", "VendorY_Binary"
  +readRawData(): String // Returns data in proprietary format
}

class Measurement {
  -measurementId: String
  -timestamp: DateTime
  -rawValue: String
  -normalizedValue: Double
  -unit: String // e.g., "Celsius", "Percent", "hPa"
  +isNormalized(): Boolean
}

class Organization {
  -organizationId: String
  -name: String
  -type: OrganizationType
}

enum OrganizationType {
  GovernmentBody
  AcademicOrganization
  InfrastructureOperator
}

class Manufacturer {
  -manufacturerId: String
  -name: String
}

class DataIngestionService {
  +ingestRawTelemetry(rawReadings: Map<Sensor, String>): List<Measurement>
}

class DataNormalizationService {
  +normalizeMeasurement(sensor: Sensor, rawValue: String): Measurement
}

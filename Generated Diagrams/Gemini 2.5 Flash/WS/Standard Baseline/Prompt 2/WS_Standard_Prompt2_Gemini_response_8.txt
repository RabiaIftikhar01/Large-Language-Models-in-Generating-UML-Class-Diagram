' Classes
class EnvironmentalMonitoringPlatform {
    - platformName: String
    + initiateDataRequest(stationId: String)
    + receiveRawTelemetry(rawMeasurement: RawMeasurement)
    + ensureDependableOperation()
}

class GeographicRegion {
    - regionId: String
    - name: String
    - coverageArea: String
}

class WeatherStation {
    - stationId: String
    - location: String
    - currentReportingInterval: Duration
    - lastTransmissionTime: DateTime
    + transmitData(): RawMeasurement[]
}

class Sensor {
    - sensorId: String
    - manufacturer: String
    - model: String
    - proprietaryFormatIdentifier: String
    + generateRawMeasurement(): RawMeasurement
}

class RawMeasurement {
    - measurementId: String
    - sensorId: String
    - timestamp: DateTime
    - rawData: byte[]
    - formatVersion: String
}

class NormalizedMeasurement {
    - measurementId: String
    - sensorId: String
    - timestamp: DateTime
    - normalizedData: Map<String, Object>
    - units: Map<String, String>
}

class DataIngestionService {
    - maxIngestionRate: Long // e.g., measurements/second
    + ingest(rawMeasurement: RawMeasurement)
    + bufferAndRoute(rawMeasurement: RawMeasurement)
}

class DataNormalizationService {
    + normalize(rawMeasurement: RawMeasurement): NormalizedMeasurement
    + registerFormat(formatId: String, strategy: NormalizationStrategy)
}

abstract class StationManager {
    - managerId: String
    - name: String
    + manageStation(station: WeatherStation)
}

class GovernmentBody extends StationManager {
    - departmentName: String
}

class AcademicOrganization extends StationManager {
    - institutionName: String
    - researchFocus: String
}

class InfrastructureOperator extends StationManager {
    - operatorType: String
    - serviceArea: String
}

interface NormalizationStrategy {
    + apply(rawData: byte[], formatVersion: String): Map<String, Object>
}

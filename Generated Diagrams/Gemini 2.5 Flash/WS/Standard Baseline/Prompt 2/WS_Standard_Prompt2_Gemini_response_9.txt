class WeatherStation {
    + stationId: String
    + name: String
    + location: String (e.g., "Lat, Lon")
    + reportingInterval: Duration
    + lastCommunicationTime: DateTime
    --
    + transmitData(): RawMeasurement[]
}

class Sensor {
    + sensorId: String
    + manufacturer: String
    + model: String
    + sensorType: String (e.g., Temperature, Humidity)
    + proprietaryFormatDetails: String
    --
    + readRawData(): RawMeasurement
}

class Organization {
    + orgId: String
    + name: String
    + type: OrgType
}

enum OrgType {
    GOVERNMENT
    ACADEMIC
    INFRASTRUCTURE_OPERATOR
}

class GeographicRegion {
    + regionId: String
    + name: String
}

class RawMeasurement {
    + rawPayload: String
    + timestamp: DateTime
    + sensorId: String
    + stationId: String
}

class NormalizedMeasurement {
    + value: Double
    + unit: String
    + timestamp: DateTime
    + sensorId: String
    + stationId: String
    + dataType: String
}

class DataIngestionModule {
    + status: IngestionStatus
    --
    + receiveData(data: RawMeasurement)
    + routeToNormalization(data: RawMeasurement)
}

enum IngestionStatus {
    ACTIVE
    PAUSED
    ERROR
}

class DataNormalizationModule {
    + normalizationRules: Map<String, String>
    --
    + normalize(data: RawMeasurement): NormalizedMeasurement
}

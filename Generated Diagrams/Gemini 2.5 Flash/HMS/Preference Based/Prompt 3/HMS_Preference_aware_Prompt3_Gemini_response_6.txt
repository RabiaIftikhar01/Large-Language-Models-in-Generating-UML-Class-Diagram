    + addCharge(charge: Charge)

    + getRawCost(): double

}

' Strategy pattern for Billing Policies
interface BillingPolicy {
    + calculateTotal(patient: Patient, charges: List<Charge>): double
}

class InsuredBillingPolicy implements BillingPolicy {
    - insurerCoverageRules: Map<String, Double> ' e.g., procedure type -> % coverage
    - coPayAmount: double
    --
    + calculateTotal(patient: Patient, charges: List<Charge>): double
}

class GovernmentBillingPolicy implements BillingPolicy {
    - subsidizedRates: Map<String, Double> ' e.g., procedure type -> fixed subsidy
    --
    + calculateTotal(patient: Patient, charges: List<Charge>): double
}

class SelfFundedBillingPolicy implements BillingPolicy {
    + calculateTotal(patient: Patient, charges: List<Charge>): double
}

' Invoice and Service for orchestration
class Invoice {
    - invoiceId: String

    - issueDate: Date

    --
    + getInvoiceId(): String
    + getTotalAmount(): double
    + getPatient(): Patient
    + getHospitalStay(): HospitalStay

class BillingService {
    --
    + createDischargeInvoice(stay: HospitalStay): Invoice
    - selectBillingPolicy(category: PatientCategory): BillingPolicy
}

' Classes for Patient and Hospital Stay
class Patient {
    - patientId : String
    - name : String
    - dateOfBirth : Date
    - billingType : BillingType <<enum>>
    --
    + getPatientId() : String
    + getName() : String
    + getBillingType() : BillingType
}

class HospitalStay {
    - stayId : String
    - admissionDate : Date
    - dischargeDate : Date
    --
    + getStayId() : String
    + getAdmissionDate() : Date
    + getDischargeDate() : Date
    + getCharges() : List<Charge>
}

enum BillingType {
    INSURANCE
    GOVERNMENT
    SELF_PAY
}

' Abstract Charge and its concrete implementations (Pricing Components)
abstract class Charge {
    - chargeId : String
    - description : String
    - originalAmount : double
    - chargeDate : Date
    --
    + getChargeId() : String
    + getDescription() : String
    + getOriginalAmount() : double
    + getChargeDate() : Date
}

class ProcedureCharge extends Charge {
    - procedureCode : String
    - procedureName : String
    --
    + getProcedureCode() : String
}

class MedicationCharge extends Charge {
    - medicationName : String
    - dosage : String
    --
    + getMedicationName() : String
}

class DiagnosticTestCharge extends Charge {
    - testType : String
    - result : String
    --
    + getTestType() : String
}

class RoomCharge extends Charge {
    - roomNumber : String
    - dailyRate : double
    - numDays : int
    --
    + getRoomNumber() : String
}

' Billing Policy (Strategy Pattern for interchangeable billing rules)
interface IBillingPolicy {
    + calculateBill(stay : HospitalStay, patient : Patient) : Bill
}

class InsurancePolicy implements IBillingPolicy {
    - coverageRules : String
    - reimbursementLimits : double
    - coPaymentStructure : String
    --
    + calculateBill(stay : HospitalStay, patient : Patient) : Bill
}

class GovernmentPolicy implements IBillingPolicy {
    - subsidizedRates : Map<String, Double>
    --
    + calculateBill(stay : HospitalStay, patient : Patient) : Bill
}

class SelfPayPolicy implements IBillingPolicy {
    - discountRate : double
    --
    + calculateBill(stay : HospitalStay, patient : Patient) : Bill
}

' Bill and BillEntry (Association Class for adjusted amounts and auditing)
class Bill {
    - billId : String
    - issueDate : Date
    - totalAmount : double
    --
    + getBillId() : String
    + getIssueDate() : Date
    + getTotalAmount() : double
    + addBillEntry(entry : BillEntry) : void
}

class BillEntry {
    - entryId : String
    - adjustedAmount : double
    - policyNotes : String
    --
    + getEntryId() : String
    + getAdjustedAmount() : double
    + getPolicyNotes() : String
}

' Billing Service (Context for the Strategy)
class BillingService {
    --
    + generateBill(patient : Patient, stay : HospitalStay) : Bill
    - getPolicyForPatient(patient : Patient) : IBillingPolicy
}

' Relationships
Patient "1" -- "0..*" HospitalStay : has

HospitalStay "1" -- "0..*" Charge : contains

Bill "1" o-- "1..*" BillEntry : includes

BillEntry "1" -- "1" Charge : references

BillingService "1" --> "1" IBillingPolicy : uses >
BillingService "1" --> "1" Patient : processes
BillingService "1" --> "1" HospitalStay : for
BillingService "1" --> "1" Bill : generates

Patient --> BillingType : has

' Explanatory notes
note "Strategy Pattern for flexible billing rules\ninterchangeable at runtime." as N1

' Treatments / Services (Pricing Elements)
abstract class Treatment {
    - id : String
    - description : String
    + {abstract} getBaseCost() : BigDecimal
    + {abstract} getDescription() : String
}

class ClinicalProcedure extends Treatment {
    - procedureCode : String
    + getBaseCost() : BigDecimal
}

class Pharmaceutical extends Treatment {
    - drugCode : String
    + getBaseCost() : BigDecimal
}

class LaboratoryDiagnostic extends Treatment {
    - testCode : String
    + getBaseCost() : BigDecimal
}

class RoomUtilization extends Treatment {
    - roomNumber : String
    - dailyRate : BigDecimal
    - numberOfDays : int
    + getBaseCost() : BigDecimal
}

' Billing Policies (Strategy Pattern)
interface BillingStrategy {
    + {abstract} getUnitPrice(treatment : Treatment, patient : Patient) : BigDecimal
    + {abstract} calculateBilledAmount(treatment : Treatment, patient : Patient, quantity : int) : BigDecimal
}

class InsuranceBillingStrategy implements BillingStrategy {
    - insurerPolicy : InsurerPolicy
    + getUnitPrice(treatment : Treatment, patient : Patient) : BigDecimal
    + calculateBilledAmount(treatment : Treatment, patient : Patient, quantity : int) : BigDecimal
}

class GovernmentBillingStrategy implements BillingStrategy {
    - subsidySchedule : SubsidySchedule
    + getUnitPrice(treatment : Treatment, patient : Patient) : BigDecimal
    + calculateBilledAmount(treatment : Treatment, patient : Patient, quantity : int) : BigDecimal
}

class SelfPayBillingStrategy implements BillingStrategy {
    + getUnitPrice(treatment : Treatment, patient : Patient) : BigDecimal
    + calculateBilledAmount(treatment : Treatment, patient : Patient, quantity : int) : BigDecimal
}

' Policy/Schedule Data (Configurable Elements)
class InsurerPolicy {
    - policyId : String
    - coverageDetails : Map<String, BigDecimal>
    - coPaymentModel : Map<String, BigDecimal>
    + getCoverageFactor(treatment : Treatment) : BigDecimal
    + getCoPayment(treatment : Treatment) : BigDecimal
}

class SubsidySchedule {
    - scheduleId : String
    - subsidizedRates : Map<String, BigDecimal>
    + getSubsidizedPrice(treatment : Treatment) : BigDecimal
}

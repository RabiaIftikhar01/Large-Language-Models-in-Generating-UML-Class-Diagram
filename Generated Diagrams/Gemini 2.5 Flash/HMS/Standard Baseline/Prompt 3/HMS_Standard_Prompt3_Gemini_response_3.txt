title Hospital Billing System - UML Class Diagram

' Abstract classes and Interfaces
abstract class BillableItem {
    + itemId: String
    + description: String
    + baseCost: decimal
    --
    + getCalculatedCost(): decimal
}

interface PricingStrategy {
    + calculatePrice(item: BillableItem): decimal
}

interface BillingPolicy {
    + calculatePatientResponsibility(stay: HospitalStay): Invoice
}

' Concrete Billable Items
class MedicalProcedure extends BillableItem {
    + procedureCode: String
}

class Medication extends BillableItem {
    + medicationCode: String
    + quantity: int
}

class DiagnosticService extends BillableItem {
    + serviceCode: String
}

class AccommodationCharge extends BillableItem {
    + roomType: String
    + days: int
}

' Concrete Pricing Strategies
class FixedPriceStrategy implements PricingStrategy {
    + fixedAmount: decimal
}

class UnitCostStrategy implements PricingStrategy {
    + unitCost: decimal
}

class TieredPriceStrategy implements PricingStrategy {
    + tiers: Map<decimal, decimal>
}

' Patient and Stay Information
class Patient {
    + patientId: String
    + name: String
    + dateOfBirth: Date
    --
    + patientCategory: PatientCategory
}

enum PatientCategory {
    INSURED
    GOVERNMENT_BENEFICIARY
    SELF_FUNDED
}

class HospitalStay {
    + stayId: String
    + admissionDate: Date
    + dischargeDate: Date
}

' Concrete Billing Policies (Strategy Implementations)
class InsuredBillingPolicy implements BillingPolicy {
    ' Uses InsurerCoverage and CoPayArrangement for calculations
}

class GovernmentBeneficiaryBillingPolicy implements BillingPolicy {
    ' Uses GovernmentSubsidy for calculations
}

class SelfFundedBillingPolicy implements BillingPolicy {
    ' Patient is responsible for the full cost
}

' Policy Configuration Data (frequently modified)
class InsurerCoverage {
    + policyNumber: String
    + coverageDetails: String
    + deductible: decimal
    + outOfPocketMax: decimal
}

class CoPayArrangement {
    + coPayAmount: decimal
    + coPayPercentage: decimal
}

class GovernmentSubsidy {
    + subsidyAmount: decimal
    + eligibilityCriteria: String
}

' Invoice and Auditing
class Invoice {
    + invoiceId: String
    + generationDate: Date
    + totalOriginalCost: decimal
    + totalPatientResponsibility: decimal
    + totalInsurerResponsibility: decimal
    + totalSubsidyApplied: decimal
}

class InvoiceLineItem {
    + originalCost: decimal
    + adjustedCost: decimal
    + reasonForAdjustment: String
}

' Orchestration / Main Service
class BillingService {
    --
    + generateInvoice(patient: Patient, stay: HospitalStay): Invoice
    - getBillingPolicy(category: PatientCategory): BillingPolicy
}

' --- Charge Sources and Pricing ---
' Design Practice 21: Identify relationships (inheritance, association)

  + getSourceId(): String
  + getName(): String

  ' ChargeSource has operations (getters) and an abstract method, so it's a valid class.
  + {abstract} getUnitPrice(): double
}

' --- Billing Core ---
class Bill {
  - billId: String
  - billDate: Date
  - totalAmount: double
  - status: String
  ' Design Practice 9: Access methods
  + getBillId(): String
  + getBillDate(): Date
  + getTotalAmount(): double
  + getStatus(): String

' --- Billing Policies (Strategy Pattern) ---
' Design Practice 27: Classes should be loosely coupled.
' The use of an interface for BillingPolicy ensures loose coupling and extensibility.
interface BillingPolicy {
  ' Design Practice 10: Implementor operation
  + applyPolicy(bill: Bill, patient: Patient): void
}

class InsuranceBillingPolicy {
  - coverageRules: String
  - reimbursementLimits: double
  - coPaymentStructure: String
  + getCoverageRules(): String
  + getReimbursementLimits(): double
  + getCoPaymentStructure(): String
  + applyPolicy(bill: Bill, patient: Patient): void
}

class GovernmentProgramBillingPolicy {
  - subsidizedRates: Map<String, Double> ' Example: Map of service codes to subsidized rates
  + getSubsidizedRates(): Map<String, Double>
  + applyPolicy(bill: Bill, patient: Patient): void
}

class SelfPayBillingPolicy {
  + applyPolicy(bill: Bill, patient: Patient): void
}

BillingPolicy <|.. InsuranceBillingPolicy
BillingPolicy <|.. GovernmentProgramBillingPolicy
BillingPolicy <|.. SelfPayBillingPolicy

  + getPatientId(): String

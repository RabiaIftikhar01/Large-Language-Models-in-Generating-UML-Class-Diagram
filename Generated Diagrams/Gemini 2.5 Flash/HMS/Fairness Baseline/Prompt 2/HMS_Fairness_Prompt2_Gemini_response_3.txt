' ----------------------------------------------------------------------------------------------------
' Treatment & Pricing Package (DP 1, 14, 21)
' Manages different types of treatments and their respective pricing strategies.
' Uses the Strategy pattern for flexible pricing logic per treatment type (DP 1, 11, 12, 27).
' ----------------------------------------------------------------------------------------------------
package "Treatment & Pricing" {
  abstract class Treatment {
    -treatmentId: String
    -description: String
    -datePerformed: Date
    -pricingStrategy: PricingStrategy
    --
    +getTreatmentId(): String
    +getDescription(): String
    +getPrice(): Double ' Delegates to PricingStrategy (DP 8, 10)
    +setPricingStrategy(strategy: PricingStrategy)
  }

  class ClinicalProcedure extends Treatment {
    -procedureCode: String
    --
    +getProcedureCode(): String
  }

  class Pharmaceutical extends Treatment {
    -drugCode: String
    -dosage: String
    --
    +getDrugCode(): String
    +getDosage(): String
  }

  class LabDiagnostic extends Treatment {
    -testCode: String
    -result: String
    --
    +getTestCode(): String
    +getResult(): String
  }

  class RoomUtilization extends Treatment {
    -roomNumber: String
    -daysOccupied: int
    --
    +getRoomNumber(): String
    +getDaysOccupied(): int
  }

  interface PricingStrategy {
    +calculatePrice(treatment: Treatment): Double
  }

  class ClinicalPricingStrategy implements PricingStrategy {
    +calculatePrice(treatment: Treatment): Double
  }

  class PharmaceuticalPricingStrategy implements PricingStrategy {
    +calculatePrice(treatment: Treatment): Double
  }

  class LabPricingStrategy implements PricingStrategy {
    +calculatePrice(treatment: Treatment): Double
  }

  class RoomPricingStrategy implements PricingStrategy {
    +calculatePrice(treatment: Treatment): Double
  }
}

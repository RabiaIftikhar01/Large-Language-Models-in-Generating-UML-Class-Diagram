title Hospital Discharge Billing System - Class Diagram

' Value Objects
class Money {
    - amount: BigDecimal
    - currency: String
    + Money(amount: BigDecimal, currency: String)
    + getAmount(): BigDecimal
    + getCurrency(): String
    + add(other: Money): Money
    + subtract(other: Money): Money
    + multiply(factor: BigDecimal): Money
}

' Enums
enum InvoiceStatus {
    PENDING
    PAID
    CANCELLED
}

enum PatientType {
    INSURED
    GOVERNMENT
    SELF_PAY
}

' Core Entities
class Patient {
    - patientId: String
    - name: String
    - dateOfBirth: Date
    - patientType: PatientType
    + getPatientId(): String
    + getName(): String
    + getDateOfBirth(): Date
    + getPatientType(): PatientType
}

class Hospitalization {
    - hospitalizationId: String
    - admissionDate: Date
    - dischargeDate: Date
    + getHospitalizationId(): String
    + getAdmissionDate(): Date
    + getDischargeDate(): Date
    + addBillableItem(item: BillableItem): void
    + getBillableItems(): List<BillableItem>
}

class Invoice {
    - invoiceId: String
    - issueDate: Date
    - totalAmount: Money
    - status: InvoiceStatus
    + getInvoiceId(): String
    + getIssueDate(): Date
    + getTotalAmount(): Money
    + getStatus(): InvoiceStatus
    + addBillingEntry(entry: BillingEntry): void
    + calculateTotalAmount(): void
}

class BillingEntry {
    - entryId: String
    - description: String
    - quantity: int
    - unitPrice: Money
    - chargedAmount: Money
    + getEntryId(): String
    + getDescription(): String
    + getQuantity(): int
    + getUnitPrice(): Money
    + getChargedAmount(): Money
}

' Billable Items Hierarchy
abstract class BillableItem {
    - itemId: String
    - description: String
    - basePrice: Money
    - quantity: int
    + getItemId(): String
    + getDescription(): String
    + getBasePrice(): Money
    + getQuantity(): int
    + calculateBaseAmount(): Money
}

class Procedure extends BillableItem {
    - procedureCode: String
    + getProcedureCode(): String
}

class Pharmaceutical extends BillableItem {
    - drugCode: String
    + getDrugCode(): String
}

class LabDiagnostic extends BillableItem {
    - testCode: String
    + getTestCode(): String
}

class RoomUtilization extends BillableItem {
    - roomNumber: String
    - days: int
    + getRoomNumber(): String
    + getDays(): int
}

' Billing Policies (Strategy Pattern)
interface IBillingPolicy {
    + applyPolicy(billableItems: List<BillableItem>, pricingService: PricingService): List<BillingEntry>
    + getPolicyName(): String
}

class InsurerBillingPolicy implements IBillingPolicy {
    - insurancePolicy: InsurancePolicy
    + InsurerBillingPolicy(policy: InsurancePolicy)
    + applyPolicy(billableItems: List<BillableItem>, pricingService: PricingService): List<BillingEntry>
    + getPolicyName(): String
}

class GovernmentBillingPolicy implements IBillingPolicy {
    + applyPolicy(billableItems: List<BillableItem>, pricingService: PricingService): List<BillingEntry>
    + getPolicyName(): String
}

class SelfPayBillingPolicy implements IBillingPolicy {
    + applyPolicy(billableItems: List<BillableItem>, pricingService: PricingService): List<BillingEntry>
    + getPolicyName(): String
}

' Policy-Related Entities
class Insurer {
    - insurerId: String
    - name: String
    + getInsurerId(): String
    + getName(): String
}

class InsurancePolicy {
    - policyId: String
    - coverageDetails: String
    - reimbursementCeilings: Map<String, Money>
    - coPaymentModels: Map<String, Money>
    + getPolicyId(): String
    + getCoverageDetails(): String
    + getReimbursementCeilings(): Map<String, Money>
    + getCoPaymentModels(): Map<String, Money>
}

' Services/Orchestration
class PricingService {
    + getBasePrice(itemId: String): Money
    + getSubsidizedPrice(itemId: String): Money
    + getInsurerSpecificPrice(itemId: String, insurerId: String): Money
}

class BillingPolicyFactory {
    + createPolicy(patientType: PatientType, patient: Patient): IBillingPolicy
}

class InvoiceGenerator {
    - pricingService: PricingService
    - policyFactory: BillingPolicyFactory
    + InvoiceGenerator(pricingService: PricingService, policyFactory: BillingPolicyFactory)
    + generateInvoice(hospitalization: Hospitalization, patient: Patient): Invoice
}

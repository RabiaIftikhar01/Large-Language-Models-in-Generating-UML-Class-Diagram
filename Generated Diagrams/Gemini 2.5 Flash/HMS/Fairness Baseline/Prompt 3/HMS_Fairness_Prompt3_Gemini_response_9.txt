  GOVERNMENT_BENEFICIARY

  PAID
  CANCELLED

  - invoiceId: String

class InvoiceLineItem {
  - lineItemId: String
  - description: String
  - quantity: int
  - unitCost: decimal
  - billedAmount: decimal
  - adjustmentAmount: decimal
  --
  + calculateBilledAmount(policy: BillingPolicy, patient: Patient, config: PolicyConfiguration): void
}

interface BillingPolicy <<Strategy>> {
  + {abstract} calculateBilledAmount(record: TreatmentRecord, patient: Patient, config: PolicyConfiguration): decimal
}

class InsuredBillingPolicy implements BillingPolicy {
  - <<private>> getCoverageParameters(patient: Patient, config: PolicyConfiguration): Map<String, String>
  --
  + calculateBilledAmount(record: TreatmentRecord, patient: Patient, config: PolicyConfiguration): decimal
}

class GovernmentBeneficiaryBillingPolicy implements BillingPolicy {
  - <<private>> getSubsidyValues(patient: Patient, config: PolicyConfiguration): Map<String, String>
  --
  + calculateBilledAmount(record: TreatmentRecord, patient: Patient, config: PolicyConfiguration): decimal
}

class SelfFundedBillingPolicy implements BillingPolicy {
  + calculateBilledAmount(record: TreatmentRecord, patient: Patient, config: PolicyConfiguration): decimal
}

class PolicyConfiguration {
  - parameters: Map<PatientCategoryType, Map<String, String>> ' e.g., { "INSURED": { "coPayRate": "0.1", "threshold": "1000" } }
  --
  + getParameter(policyType: PatientCategoryType, key: String): String
  + updateParameter(policyType: PatientCategoryType, key: String, value: String): void
}

class InvoiceCalculator {
  - billingPolicies: Map<PatientCategoryType, BillingPolicy>
  - policyConfig: PolicyConfiguration
  --
  + createInvoice(patient: Patient): DischargeInvoice
  - determineBillingPolicy(patient: Patient): BillingPolicy
}

' Explanatory Notes
note left of BillingPolicy
  This interface defines the strategy
  for calculating billed amounts based
  on patient category and policy rules.
  Ensures dynamic interchangeability
  at runtime as per requirements.
end note

note right of PolicyConfiguration
  Stores configurable parameters
  (e.g., coverage, reimbursement, co-pay, subsidy values)
  for billing policies. Allows dynamic updates
  without triggering structural redesign.
end note

note "The InvoiceCalculator orchestrates the invoice generation process.\nIt selects the appropriate BillingPolicy based on the Patient's category\nand utilizes PolicyConfiguration for dynamic rule parameters, ensuring\nloose coupling and extensibility." as N1
InvoiceCalculator .. N1

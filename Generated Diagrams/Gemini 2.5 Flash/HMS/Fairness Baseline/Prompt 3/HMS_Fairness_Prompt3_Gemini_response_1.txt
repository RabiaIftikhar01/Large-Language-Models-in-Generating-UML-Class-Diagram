' Note on overall system architecture and design principles
note "This model emphasizes strict separation of concerns:\n- Patient Information\n- Cost Aggregation & Pricing\n- Billing Policies\n\nIt leverages the Strategy Pattern for dynamic interchangeability\nof billing and pricing rules, ensuring extensibility and maintainability." as N_Architecture

' Package for Patient Information
package "Patient Information" {
    class Patient {
        - patientId: String
        - name: String
        - dateOfBirth: Date
        - admissionDate: Date
        - dischargeDate: Date
        - patientCategory: PatientCategory
        --
        + getPatientId(): String
        + getName(): String
        + getPatientCategory(): PatientCategory
    }

    enum PatientCategory {
        INSURED
        GOVERNMENT_BENEFICIARY
        SELF_FUNDED
    }
    Patient "1" --> "1" PatientCategory : has
}

' Package for Cost Aggregation and Pricing Logic
package "Cost Aggregation & Pricing" {
    class Treatment {
        - treatmentId: String
        - description: String
        - datePerformed: Date
        --
        + getTreatmentId(): String
        + getDescription(): String
        + getDatePerformed(): Date
        + addCostItem(item: CostItem): void
        + getCostItems(): List<CostItem>
    }

    abstract class CostItem {
        - itemId: String
        - description: String
        - quantity: int
        - baseUnitPrice: BigDecimal
        # pricingRule: PricingRule  ' Specific rule for this type of cost item
        --
        + getItemId(): String
        + getDescription(): String
        + getQuantity(): int
        + getBaseUnitPrice(): BigDecimal
        + calculateBaseCost(): BigDecimal
    }

    class MedicalProcedure extends CostItem {
        + calculateBaseCost(): BigDecimal
    }

    class Medication extends CostItem {
        + calculateBaseCost(): BigDecimal
    }

    class DiagnosticService extends CostItem {
        + calculateBaseCost(): BigDecimal
    }

    class AccommodationCharge extends CostItem {
        - numberOfNights: int
        --
        + getNumberOfNights(): int
        + calculateBaseCost(): BigDecimal
    }

    interface PricingRule {
        + apply(costItem: CostItem): BigDecimal
    }

    class ProcedurePricingRule implements PricingRule {
        + apply(costItem: CostItem): BigDecimal
    }

    class MedicationPricingRule implements PricingRule {
        + apply(costItem: CostItem): BigDecimal
    }

    class DiagnosticPricingRule implements PricingRule {
        + apply(costItem: CostItem): BigDecimal
    }

    class AccommodationPricingRule implements PricingRule {
        + apply(costItem: CostItem): BigDecimal
    }

    Treatment "1" o-- "1..*" CostItem : includes
    CostItem "1" *-- "1" PricingRule : uses
    MedicalProcedure "1" --> "1" ProcedurePricingRule : <<binds to>>
    Medication "1" --> "1" MedicationPricingRule : <<binds to>>
    DiagnosticService "1" --> "1" DiagnosticPricingRule : <<binds to>>
    AccommodationCharge "1" --> "1" AccommodationPricingRule : <<binds to>>

    note "Each CostItem type is configured with a specific PricingRule\nfor its base cost calculation, enabling distinct pricing logic." as N_PricingRule
    CostItem . N_PricingRule
}

' Package for Billing Policies (Strategy Pattern implementation)
package "Billing Policies" {
    interface BillingPolicy {
        + calculateBilledAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal): BigDecimal
        + getAdjustmentAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal, billedAmount: BigDecimal): BigDecimal
    }

    class InsuredPatientBillingPolicy implements BillingPolicy {
        - coverageThreshold: BigDecimal
        - coPayPercentage: BigDecimal
        - reimbursementThreshold: BigDecimal
        --
        + calculateBilledAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal): BigDecimal
        + getAdjustmentAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal, billedAmount: BigDecimal): BigDecimal
    }

    class GovernmentBeneficiaryBillingPolicy implements BillingPolicy {
        - subsidizedAmount: BigDecimal
        --
        + calculateBilledAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal): BigDecimal
        + getAdjustmentAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal, billedAmount: BigDecimal): BigDecimal
    }

    class SelfFundedPatientBillingPolicy implements BillingPolicy {
        + calculateBilledAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal): BigDecimal
        + getAdjustmentAmount(patient: Patient, costItem: CostItem, baseCost: BigDecimal, billedAmount: BigDecimal): BigDecimal
    }

    note "Billing policies are dynamically interchangeable at runtime\nbased on patient category and regulatory directives (Strategy Pattern)." as N_BillingPolicy
    BillingPolicy . N_BillingPolicy
}

' Package for Patient Information
package "Patient Information" {
    class Patient {
        - patientId: String
        - name: String
        - dateOfBirth: Date
        - patientCategory: PatientCategory
        --
        + getPatientInfo(): Map<String, String>
    }

    enum PatientCategory {
        INSURED
        GOVERNMENT_BENEFICIARY
        SELF_FUNDED
    }

    Patient "1" -- "1" PatientCategory : has
}

' Package for Healthcare Services and Treatments
package "Healthcare Services" {
    class HospitalStay {
        - stayId: String
        - admissionDate: Date
        - dischargeDate: Date
        --
        + addTreatment(treatment: Treatment): void
        + getTreatments(): List<Treatment>
    }

    abstract class Treatment {
        - treatmentId: String
        - description: String
        - datePerformed: Date
        - baseCost: BigDecimal
        --
        + getBaseCost(): BigDecimal
    }

    class MedicalProcedure extends Treatment {
        - procedureCode: String
    }

    class Medication extends Treatment {
        - drugName: String
        - dosage: String
        - quantity: int
    }

    class DiagnosticService extends Treatment {
        - serviceCode: String
        - result: String
    }

    class AccommodationCharge extends Treatment {
        - roomNumber: String
        - dailyRate: BigDecimal
        - numberOfDays: int
    }

    HospitalStay "1" *-- "1..*" Treatment : includes
}

' Package for Billing Policies (Strategy Pattern)
package "Billing Policies" {
    interface BillingPolicy {
        + applyPolicy(invoice: Invoice): void
    }

    class InsuredBillingPolicy implements BillingPolicy {
        - coverageParameters: Map<String, BigDecimal>
        - reimbursementThresholds: Map<String, BigDecimal>
        - coPayArrangements: Map<String, BigDecimal>
        --
        + applyPolicy(invoice: Invoice): void
    }

    class GovernmentBeneficiaryBillingPolicy implements BillingPolicy {
        - subsidyValues: Map<String, BigDecimal>
        --
        + applyPolicy(invoice: Invoice): void
    }

    class SelfFundedBillingPolicy implements BillingPolicy {
        - administrativeFeeRate: BigDecimal
        --
        + applyPolicy(invoice: Invoice): void
    }

    class BillingPolicyFactory {
        - policies: Map<PatientCategory, BillingPolicy>
        --
        + getPolicy(patientCategory: PatientCategory): BillingPolicy
    }

    BillingPolicyFactory "1" --> "1..*" BillingPolicy : creates
    BillingPolicyFactory "1" --> "1" PatientCategory : uses
}

' Package for Invoice Generation and Orchestration
package "Invoice Generation" {
    class Invoice {
        - invoiceId: String
        - issueDate: Date
        - totalAmount: BigDecimal
        - patient: Patient
        - hospitalStay: HospitalStay
        - billedItems: List<BilledItem>
        --
        + addBilledItem(item: BilledItem): void
        + calculateTotal(): void
        + getBilledItems(): List<BilledItem>
    }

    class BilledItem {
        - billedItemId: String
        - description: String
        - originalCost: BigDecimal
        - adjustedCost: BigDecimal
        - treatmentRef: Treatment
        --
        + setAdjustedCost(cost: BigDecimal): void
        + getFinalCost(): BigDecimal
    }

    class DischargeInvoiceGenerator {
        - pricingService: PricingService
        - billingPolicyFactory: BillingPolicyFactory
        --
        + generateInvoice(patient: Patient, stay: HospitalStay): Invoice
        - selectBillingPolicy(patientCategory: PatientCategory): BillingPolicy
    }

    Invoice "1" *-- "1..*" BilledItem : contains
    BilledItem "0..*" --> "1" Treatment : references

    Invoice "1" -- "1" Patient : for
    Invoice "1" -- "1" HospitalStay : covers

    DischargeInvoiceGenerator "1" --> "1" PricingService : uses

    DischargeInvoiceGenerator "1" --> "1" Invoice : generates

    DischargeInvoiceGenerator "1" --> "1" HospitalStay : processes
}
